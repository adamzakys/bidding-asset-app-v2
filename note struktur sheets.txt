A2 pada sheet/tabel 4 itu:
=ARRAYFORMULA(
  LET(
    _bids_valid, IFERROR(FILTER(tb_bids!A2:F, tb_bids!F2:F = "VALID"), {"","","","","",""}),
    _asset_ids, INDEX(_bids_valid,, 3),
    _nominal_bids, INDEX(_bids_valid,, 5),
    _emails, INDEX(_bids_valid,, 4),
    
    _max_bids, MAP(tb_assets!A2:A, LAMBDA(_id, 
      IF(_id = "", "", IFERROR(MAX(FILTER(_nominal_bids, _asset_ids = _id)), 0))
    )),
    
    _winner_emails, MAP(tb_assets!A2:A, _max_bids, LAMBDA(_id, _max, 
      IF(_id = "", "", IFERROR(INDEX(FILTER(_emails, (_asset_ids = _id) * (_nominal_bids = _max)), 1), "-"))
    )),
    
    _winner_names, IFERROR(VLOOKUP(_winner_emails, tb_users!A:D, 2, FALSE), "-"),
    _winner_nips, IFERROR(VLOOKUP(_winner_emails, tb_users!A:C, 3, FALSE), "-"),
    _winner_wa, IFERROR(VLOOKUP(_winner_emails, tb_users!A:D, 4, FALSE), "-"),
    
    _sisa_waktu, MAP(tb_assets!H2:H, LAMBDA(_deadline,
      IF(_deadline = "", "", 
        IF(_deadline < NOW(), 
          "CLOSED", 
          LET(
            _diff, _deadline - NOW(),
            _d, INT(_diff),
            _h, INT(MOD(_diff * 24, 24)),
            _m, INT(MOD(_diff * 1440, 60)),
            IF(_d > 0, _d & " hari ", "") & _h & " jam " & _m & " menit"
          )
        )
      )
    )),
    
    _result, FILTER(
      {tb_assets!A2:A, tb_assets!C2:C, tb_assets!B2:B, tb_assets!F2:F, _max_bids, _winner_emails, _winner_names, _winner_nips, _winner_wa, _sisa_waktu},
      tb_assets!A2:A <> ""
    ),
    
    _result
  )
)

5. Tabel view_bidder_rank

E1: Jumlah Bid
F1: Total Nominal Bid

A2 pada sheet/tabel 5 itu:
=ARRAYFORMULA(
  LET(
    _bids_valid, IFERROR(FILTER(tb_bids!D2:E, tb_bids!F2:F = "VALID"), {"",""}),
    _bidder_emails, INDEX(_bids_valid,, 1),
    _bidder_nominals, INDEX(_bids_valid,, 2),
    
    _unique_emails, UNIQUE(_bidder_emails),
    
    _bid_counts, MAP(_unique_emails, LAMBDA(_email,
      IF(_email = "", 0, COUNTIF(_bidder_emails, _email))
    )),
    
    _total_spent, MAP(_unique_emails, LAMBDA(_email,
      IF(_email = "", 0, SUMIF(_bidder_emails, _email, _bidder_nominals))
    )),
    
    _names, IFERROR(VLOOKUP(_unique_emails, tb_users!A:D, 2, FALSE), "-"),
    _nips, IFERROR(VLOOKUP(_unique_emails, tb_users!A:D, 3, FALSE), "-"),
    _wa, IFERROR(VLOOKUP(_unique_emails, tb_users!A:D, 4, FALSE), "-"),
    
    _rekap, { _unique_emails, _names, _nips, _wa, _bid_counts, _total_spent },
    _sorted, SORT(_rekap, 5, FALSE),
    
    _sorted
  )
)
